# Ranges rather than exact pins, so one requirements file resolves on every
# supported interpreter. The newest releases of pandas, numpy and scikit-learn
# require Python 3.11; pinning those exactly made the whole application
# uninstallable on 3.10 for no functional reason.
#
# Verified: Python 3.10.20 (pandas 2.3.3 / numpy 2.2.6 / scikit-learn 1.7.2)
#           Python 3.11.15 (pandas 3.0.5 / numpy 2.4.6 / scikit-learn 1.9.0)
# Full test suite passes on both.
#
# Floors are set where a feature this application actually uses was introduced;
# ceilings stop an unreleased major from silently changing behaviour.

fastapi>=0.110,<1
uvicorn>=0.27,<1
python-multipart>=0.0.9
pydantic>=2.7,<3
